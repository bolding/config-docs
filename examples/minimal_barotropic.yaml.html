# A minimal, self-contained pyGETM config -- no external data files needed, runs
# in a couple of seconds. This is the exact config used to verify the loader
# end-to-end during development (see PLAN.md's addendum).
#
# Run it:
#   conda activate pygetm   # needs pygetm -- this actually executes a simulation
#   cd pygetm-config
#   pygetm-config run examples/minimal_barotropic.yaml --stop 2020-01-01T00:05:00
#
# Or as a dry run (build + start, don't advance):
#   pygetm-config run examples/minimal_barotropic.yaml --dry-run
#
# A tiny 5x5 Cartesian grid, flat 5 m depth, barotropic 2D (fastest/simplest
# runtype -- no temperature/salinity/FABM), prescribed (constant, zero) surface
# fluxes, one in-memory output recorder capturing surface elevation.

domain:
  method: create_cartesian
  x: [0.0, 100.0, 200.0, 300.0, 400.0]   # m
  y: [0.0, 100.0, 200.0, 300.0, 400.0]   # m
  H: 5.0                                  # flat 5 m depth everywhere
  f: 0.0001                               # Coriolis parameter (rad/s), ~mid-latitude

simulation:
  runtype: 1        # BAROTROPIC_2D -- no temperature/salinity/FABM needed
  airsea:
    type: Fluxes     # prescribed (constant) surface fluxes, not calculated from meteo
    # all of taux/tauy/sp/shf/swr/pe default to 0 -- a quiescent atmosphere

runtime:
  time: "2020-01-01T00:00:00"   # start time
  timestep: 10.0                 # seconds
  split_factor: 1

output:
  files:
    - type: memory                          # in-memory recorder, no file written
      variable_requests:
        - fields: [zt]                       # surface elevation
